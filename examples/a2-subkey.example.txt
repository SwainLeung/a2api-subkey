# A2 网关凭据载体 —— 占位符示例，不含任何真实凭据
#
# 用法：复制本文件到仓库之外的本地只读目录，重命名后填入真值。
#       （例如 ~/.config/a2/a2-subkey.txt，或任意你控制的私有路径）
# 铁律：这个文件永远不要提交、不要上传、不要贴进聊天记录。

# 必填。只写主机：不要 http://、不要端口、不要结尾斜杠。
ip = <host>

# 选填，默认 3000。
port = 3000

# 必填。真子 Key：maki_ + 32 位小写十六进制，总长 37。
# 注意 maki_888*** 这种是文档里的掩码，不是真 Key —— 写进去会被诊断器判 SK-003-MASKED。
subkey = maki_<32 hex>

# 选填。设备 / 用途，用于区分多把 Key。
label = laptop-codex
